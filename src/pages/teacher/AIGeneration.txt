import React, { useEffect, useState } from 'react';
import {
  Sparkles,
  BookOpen,
  FileText,
  GraduationCap,
  ClipboardList,
  Presentation,
  BarChart3,
  Target,
  MessageSquare,
  Wand2,
  Loader2,
  CheckCircle2,
  Download,
  Lightbulb,
  Award,
  TrendingUp,
  Users,
} from 'lucide-react';
import { teacherService, type BackendTeacherCourse, type BackendTeacherChapter, type BackendTeacherLecture } from '../../services/teacher.service';
import toast from 'react-hot-toast';

const AIGeneration: React.FC = () => {
  const [courses, setCourses] = useState<BackendTeacherCourse[]>([]);
  const [selectedCourseId, setSelectedCourseId] = useState<string>('');
  const [chapters, setChapters] = useState<BackendTeacherChapter[]>([]);
  const [selectedChapterId, setSelectedChapterId] = useState<string>('');
  const [loading, setLoading] = useState(false);
  const [activeTab, setActiveTab] = useState<string>('outline');

  // Form states
  const [outlineForm, setOutlineForm] = useState({
    title: '',
    description: '',
    targetAudience: '',
    duration: '',
    learningObjectives: [] as string[],
  });

  const [contentForm, setContentForm] = useState({
    lectureId: '',
    prompt: '',
    type: 'summary' as 'summary' | 'explanation' | 'examples' | 'key_points',
  });

  const [guideForm, setGuideForm] = useState({
    lectureId: '',
    classDuration: 60,
    classSize: 20,
    teachingMode: 'offline' as 'offline' | 'online' | 'hybrid',
    studentLevel: 'intermediate' as 'beginner' | 'intermediate' | 'advanced',
  });

  const [examForm, setExamForm] = useState({
    coverage: 'course' as 'course' | 'chapter' | 'lecture',
    chapterId: '',
    lectureId: '',
    duration: 60,
    questionCount: 20,
    difficulty: 'mixed' as 'easy' | 'medium' | 'hard' | 'mixed',
    includeAnswerKey: true,
  });

  const [materialsForm, setMaterialsForm] = useState({
    chapterId: '',
    lectureId: '',
    type: 'slides' as 'slides' | 'handout' | 'worksheet' | 'all',
    format: 'pdf' as 'pptx' | 'pdf' | 'docx',
  });

  const [feedbackForm, setFeedbackForm] = useState({
    studentId: '',
    type: 'general' as 'general' | 'specific' | 'improvement',
    context: '',
  });

  // Results
  const [generatedOutline, setGeneratedOutline] = useState<any>(null);
  const [generatedContent, setGeneratedContent] = useState<any>(null);
  const [generatedGuide, setGeneratedGuide] = useState<any>(null);
  const [generatedExercises, setGeneratedExercises] = useState<any>(null);
  const [generatedExam, setGeneratedExam] = useState<any>(null);
  const [generatedMaterials, setGeneratedMaterials] = useState<any>(null);
  const [contentQuality, setContentQuality] = useState<any>(null);
  const [courseAnalytics, setCourseAnalytics] = useState<any>(null);
  const [courseDifficulty, setCourseDifficulty] = useState<any>(null);
  const [studentFeedback, setStudentFeedback] = useState<any>(null);

  useEffect(() => {
    loadCourses();
  }, []);

  const loadCourses = async () => {
    try {
      const myCourses = await teacherService.listMyCourses();
      setCourses(myCourses);
      if (myCourses.length > 0) {
        setSelectedCourseId(String(myCourses[0].id));
      }
    } catch (err: any) {
      toast.error(err?.message || 'Không thể tải danh sách khóa học');
    }
  };

  // Load chapters when course is selected (for tabs that need lecture selection)
  useEffect(() => {
    if (selectedCourseId && (activeTab === 'guide' || activeTab === 'content')) {
      loadCourseChapters(selectedCourseId);
    } else {
      setChapters([]);
      setSelectedChapterId('');
      setGuideForm(prev => ({ ...prev, lectureId: '' }));
      setContentForm(prev => ({ ...prev, lectureId: '' }));
    }
  }, [selectedCourseId, activeTab]);

  const loadCourseChapters = async (courseId: string) => {
    try {
      const content = await teacherService.getCourseContent(courseId);
      setChapters(content.chapters || []);
    } catch (err: any) {
      toast.error('Không thể tải danh sách chương/bài học');
      setChapters([]);
    }
  };

  // Generate Course Outline
  const handleGenerateOutline = async () => {
    if (!outlineForm.title.trim()) {
      toast.error('Vui lòng nhập tiêu đề khóa học');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.generateAndSaveCourseOutline({
        ...outlineForm,
        saveToDatabase: true,
      });
      setGeneratedOutline(result);
      toast.success('Đã tạo outline khóa học thành công!');
      loadCourses();
    } catch (err: any) {
      toast.error(err?.message || 'Không thể tạo outline');
    } finally {
      setLoading(false);
    }
  };

  // Generate Content
  const handleGenerateContent = async () => {
    if (!contentForm.lectureId) {
      toast.error('Vui lòng chọn bài học');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.generateContent(contentForm);
      setGeneratedContent(result);
      toast.success('Đã tạo nội dung thành công!');
    } catch (err: any) {
      toast.error(err?.message || 'Không thể tạo nội dung');
    } finally {
      setLoading(false);
    }
  };

  // Generate Teaching Guide
  const handleGenerateGuide = async () => {
    if (!guideForm.lectureId) {
      toast.error('Vui lòng chọn chương và bài học');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.generateTeachingGuide({
        lectureId: guideForm.lectureId,
        classDuration: guideForm.classDuration,
        classSize: guideForm.classSize,
        teachingMode: guideForm.teachingMode,
        studentLevel: guideForm.studentLevel,
      });
      setGeneratedGuide(result);
      toast.success('Đã tạo giáo án thành công!');
    } catch (err: any) {
      toast.error(err?.message || 'Không thể tạo giáo án');
    } finally {
      setLoading(false);
    }
  };

  // Generate Exercises
  const handleGenerateExercises = async () => {
    if (!selectedCourseId) {
      toast.error('Vui lòng chọn khóa học');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.generateExercises({
        courseId: selectedCourseId,
        count: 10,
        difficulty: 'medium',
        type: 'practice',
      });
      setGeneratedExercises(result);
      toast.success('Đã tạo bài tập thành công!');
    } catch (err: any) {
      toast.error(err?.message || 'Không thể tạo bài tập');
    } finally {
      setLoading(false);
    }
  };

  // Generate Exam
  const handleGenerateExam = async () => {
    if (!selectedCourseId) {
      toast.error('Vui lòng chọn khóa học');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.generateExam({
        courseId: selectedCourseId,
        ...examForm,
      });
      setGeneratedExam(result);
      toast.success('Đã tạo đề thi thành công!');
    } catch (err: any) {
      toast.error(err?.message || 'Không thể tạo đề thi');
    } finally {
      setLoading(false);
    }
  };

  // Generate Materials
  const handleGenerateMaterials = async () => {
    if (!selectedCourseId) {
      toast.error('Vui lòng chọn khóa học');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.generateTeachingMaterials({
        courseId: selectedCourseId,
        ...materialsForm,
      });
      setGeneratedMaterials(result);
      toast.success('Đã tạo tài liệu giảng dạy thành công!');
    } catch (err: any) {
      toast.error(err?.message || 'Không thể tạo tài liệu');
    } finally {
      setLoading(false);
    }
  };

  // Analyze Content Quality
  const handleAnalyzeQuality = async () => {
    if (!selectedCourseId) {
      toast.error('Vui lòng chọn khóa học');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.analyzeContentQuality({
        courseId: selectedCourseId,
      });
      setContentQuality(result);
      toast.success('Phân tích chất lượng hoàn tất!');
    } catch (err: any) {
      toast.error(err?.message || 'Không thể phân tích chất lượng');
    } finally {
      setLoading(false);
    }
  };

  // Get Course Analytics
  const handleGetAnalytics = async () => {
    if (!selectedCourseId) {
      toast.error('Vui lòng chọn khóa học');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.getCourseAnalytics(selectedCourseId);
      setCourseAnalytics(result);
      toast.success('Đã tải analytics thành công!');
    } catch (err: any) {
      toast.error(err?.message || 'Không thể tải analytics');
    } finally {
      setLoading(false);
    }
  };

  // Analyze Course Difficulty
  const handleAnalyzeDifficulty = async () => {
    if (!selectedCourseId) {
      toast.error('Vui lòng chọn khóa học');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.analyzeCourseDifficulty(selectedCourseId);
      setCourseDifficulty(result);
      toast.success('Phân tích độ khó hoàn tất!');
    } catch (err: any) {
      toast.error(err?.message || 'Không thể phân tích độ khó');
    } finally {
      setLoading(false);
    }
  };

  // Generate Student Feedback
  const handleGenerateFeedback = async () => {
    if (!selectedCourseId || !feedbackForm.studentId) {
      toast.error('Vui lòng chọn khóa học và học sinh');
      return;
    }
    try {
      setLoading(true);
      const result = await teacherService.generateStudentFeedback({
        courseId: selectedCourseId,
        ...feedbackForm,
      });
      setStudentFeedback(result);
      toast.success('Đã tạo feedback thành công!');
    } catch (err: any) {
      toast.error(err?.message || 'Không thể tạo feedback');
    } finally {
      setLoading(false);
    }
  };

  const tabs = [
    { id: 'outline', label: 'Tạo Outline', icon: BookOpen, color: 'bg-blue-500' },
    { id: 'content', label: 'Tạo Nội Dung', icon: FileText, color: 'bg-emerald-500' },
    { id: 'guide', label: 'Giáo Án', icon: GraduationCap, color: 'bg-amber-500' },
    { id: 'exercises', label: 'Bài Tập', icon: ClipboardList, color: 'bg-rose-500' },
    { id: 'exam', label: 'Đề Thi', icon: Award, color: 'bg-purple-500' },
    { id: 'materials', label: 'Tài Liệu', icon: Presentation, color: 'bg-indigo-500' },
    { id: 'analytics', label: 'Phân Tích', icon: BarChart3, color: 'bg-cyan-500' },
    { id: 'feedback', label: 'Feedback', icon: MessageSquare, color: 'bg-pink-500' },
  ];

  return (
    <div className="w-full pb-20 px-2 lg:px-6">
      <div className="max-w-7xl mx-auto">
        {/* Header */}
        <div className="flex flex-col lg:flex-row lg:items-end justify-between gap-6 mb-12">
          <div>
            <h1 className="text-3xl font-bold text-gray-900 flex items-center gap-3">
              <div className="p-3 bg-purple-600 rounded-2xl text-white shadow-lg shadow-purple-200">
                <Sparkles size={28} />
              </div>
              AI Generation Hub
            </h1>
            <p className="text-gray-500 font-medium text-lg leading-relaxed max-w-lg mt-4">
              Sử dụng AI để tạo nội dung, phân tích và tối ưu hóa khóa học của bạn
            </p>
          </div>

          {/* Course Selector */}
          <div className="bg-white p-4 rounded-2xl border border-gray-100 shadow-sm">
            <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
              Khóa học
            </label>
            <select
              value={selectedCourseId}
              onChange={(e) => setSelectedCourseId(e.target.value)}
              className="w-64 bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 text-sm font-bold text-gray-800 focus:outline-none focus:border-purple-500 cursor-pointer"
            >
              <option value="">Chọn khóa học...</option>
              {courses.map((c) => (
                <option key={c.id} value={c.id}>{c.title}</option>
              ))}
            </select>
          </div>
        </div>

        {/* Tabs */}
        <div className="grid grid-cols-4 md:grid-cols-8 gap-3 mb-8">
          {tabs.map((tab) => (
            <button
              key={tab.id}
              onClick={() => setActiveTab(tab.id)}
              className={`flex flex-col items-center gap-2 p-4 rounded-2xl transition-all ${
                activeTab === tab.id
                  ? 'bg-gray-900 text-white shadow-xl scale-105'
                  : 'bg-white text-gray-500 hover:bg-gray-50 border border-gray-100'
              }`}
            >
              <div className={`p-2 rounded-xl ${tab.color} text-white`}>
                <tab.icon size={18} />
              </div>
              <span className="text-[10px] font-bold uppercase tracking-wider">{tab.label}</span>
            </button>
          ))}
        </div>

        {/* Content */}
        <div className="bg-white rounded-[32px] border border-gray-100 shadow-sm p-8">
          {/* Outline Tab */}
          {activeTab === 'outline' && (
            <div className="space-y-6">
              <div className="flex items-center gap-3 mb-6">
                <div className="p-3 bg-blue-500 rounded-xl text-white">
                  <BookOpen size={24} />
                </div>
                <div>
                  <h2 className="text-xl font-bold text-gray-900">Tạo Outline Khóa Học</h2>
                  <p className="text-sm text-gray-400">AI sẽ tạo cấu trúc khóa học hoàn chỉnh</p>
                </div>
              </div>

              <div className="space-y-4">
                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    Tiêu đề khóa học
                  </label>
                  <input
                    value={outlineForm.title}
                    onChange={(e) => setOutlineForm({ ...outlineForm, title: e.target.value })}
                    placeholder="VD: Tiếng Anh giao tiếp cho người mới bắt đầu"
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-blue-500"
                  />
                </div>

                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    Mô tả
                  </label>
                  <textarea
                    value={outlineForm.description}
                    onChange={(e) => setOutlineForm({ ...outlineForm, description: e.target.value })}
                    placeholder="Mô tả ngắn về khóa học..."
                    rows={3}
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-blue-500"
                  />
                </div>

                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                      Đối tượng
                    </label>
                    <input
                      value={outlineForm.targetAudience}
                      onChange={(e) => setOutlineForm({ ...outlineForm, targetAudience: e.target.value })}
                      placeholder="VD: Sinh viên, người đi làm"
                      className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-blue-500"
                    />
                  </div>
                  <div>
                    <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                      Thời lượng
                    </label>
                    <input
                      value={outlineForm.duration}
                      onChange={(e) => setOutlineForm({ ...outlineForm, duration: e.target.value })}
                      placeholder="VD: 8 tuần"
                      className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-blue-500"
                    />
                  </div>
                </div>

                <button
                  onClick={handleGenerateOutline}
                  disabled={loading}
                  className="w-full py-4 bg-blue-500 text-white rounded-xl font-bold hover:bg-blue-600 transition-all flex items-center justify-center gap-2 disabled:opacity-50"
                >
                  {loading ? <Loader2 size={20} className="animate-spin" /> : <Wand2 size={20} />}
                  Tạo Outline với AI
                </button>

                {generatedOutline && (
                  <div className="mt-6 p-6 bg-blue-50 rounded-2xl border border-blue-100">
                    <h3 className="font-bold text-blue-900 mb-4 flex items-center gap-2">
                      <CheckCircle2 size={20} />
                      Đã tạo outline thành công!
                    </h3>
                    <pre className="text-sm text-blue-800 whitespace-pre-wrap overflow-auto max-h-60">
                      {JSON.stringify(generatedOutline, null, 2)}
                    </pre>
                  </div>
                )}
              </div>
            </div>
          )}

          {/* Content Tab */}
          {activeTab === 'content' && (
            <div className="space-y-6">
              <div className="flex items-center gap-3 mb-6">
                <div className="p-3 bg-emerald-500 rounded-xl text-white">
                  <FileText size={24} />
                </div>
                <div>
                  <h2 className="text-xl font-bold text-gray-900">Tạo Nội Dung Bài Học</h2>
                  <p className="text-sm text-gray-400">AI sẽ tạo nội dung chi tiết cho bài học</p>
                </div>
              </div>

              <div className="space-y-4">
                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    ID Bài học
                  </label>
                  <input
                    value={contentForm.lectureId}
                    onChange={(e) => setContentForm({ ...contentForm, lectureId: e.target.value })}
                    placeholder="Nhập ID bài học"
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-emerald-500"
                  />
                </div>

                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    Loại nội dung
                  </label>
                  <select
                    value={contentForm.type}
                    onChange={(e) => setContentForm({ ...contentForm, type: e.target.value as any })}
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-emerald-500"
                  >
                    <option value="summary">Tóm tắt</option>
                    <option value="explanation">Giải thích</option>
                    <option value="examples">Ví dụ</option>
                    <option value="key_points">Điểm chính</option>
                  </select>
                </div>

                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    Yêu cầu thêm (tùy chọn)
                  </label>
                  <textarea
                    value={contentForm.prompt}
                    onChange={(e) => setContentForm({ ...contentForm, prompt: e.target.value })}
                    placeholder="Mô tả chi tiết nội dung bạn muốn..."
                    rows={3}
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-emerald-500"
                  />
                </div>

                <button
                  onClick={handleGenerateContent}
                  disabled={loading}
                  className="w-full py-4 bg-emerald-500 text-white rounded-xl font-bold hover:bg-emerald-600 transition-all flex items-center justify-center gap-2 disabled:opacity-50"
                >
                  {loading ? <Loader2 size={20} className="animate-spin" /> : <Wand2 size={20} />}
                  Tạo Nội Dung với AI
                </button>

                {generatedContent && (
                  <div className="mt-6 p-6 bg-emerald-50 rounded-2xl border border-emerald-100">
                    <h3 className="font-bold text-emerald-900 mb-4 flex items-center gap-2">
                      <CheckCircle2 size={20} />
                      Nội dung đã tạo
                    </h3>
                    <div className="prose prose-emerald max-w-none">
                      <p className="text-emerald-800 whitespace-pre-wrap">{generatedContent.content}</p>
                    </div>
                  </div>
                )}
              </div>
            </div>
          )}

          {/* Guide Tab */}
          {activeTab === 'guide' && (
            <div className="space-y-6">
              <div className="flex items-center gap-3 mb-6">
                <div className="p-3 bg-amber-500 rounded-xl text-white">
                  <GraduationCap size={24} />
                </div>
                <div>
                  <h2 className="text-xl font-bold text-gray-900">Tạo Giáo Án</h2>
                  <p className="text-sm text-gray-400">AI sẽ tạo giáo án chi tiết cho buổi học</p>
                </div>
              </div>

              <div className="space-y-4">
                {/* Chapter Selection */}
                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    Chọn Chương *
                  </label>
                  <select
                    value={selectedChapterId}
                    onChange={(e) => {
                      setSelectedChapterId(e.target.value);
                      setGuideForm(prev => ({ ...prev, lectureId: '' }));
                    }}
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-amber-500"
                  >
                    <option value="">-- Chọn chương --</option>
                    {chapters.map((chapter) => (
                      <option key={chapter.id} value={chapter.id}>
                        {chapter.title}
                      </option>
                    ))}
                  </select>
                </div>

                {/* Lecture Selection */}
                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    Chọn Bài học *
                  </label>
                  <select
                    value={guideForm.lectureId}
                    onChange={(e) => setGuideForm({ ...guideForm, lectureId: e.target.value })}
                    disabled={!selectedChapterId}
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-amber-500 disabled:opacity-50"
                  >
                    <option value="">-- Chọn bài học --</option>
                    {selectedChapterId && chapters
                      .find(c => String(c.id) === selectedChapterId)?.Lectures?.map((lecture: BackendTeacherLecture) => (
                        <option key={lecture.id} value={lecture.id}>
                          {lecture.title}
                        </option>
                      ))}
                  </select>
                  {selectedChapterId && chapters.find(c => String(c.id) === selectedChapterId)?.Lectures?.length === 0 && (
                    <p className="text-xs text-amber-600 mt-1">Chương này chưa có bài học nào</p>
                  )}
                </div>

                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                      Thời lượng (phút)
                    </label>
                    <input
                      type="number"
                      min={15}
                      max={180}
                      value={guideForm.classDuration}
                      onChange={(e) => setGuideForm({ ...guideForm, classDuration: Number(e.target.value) })}
                      className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-amber-500"
                    />
                  </div>
                  <div>
                    <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                      Số học sinh
                    </label>
                    <input
                      type="number"
                      min={1}
                      max={200}
                      value={guideForm.classSize}
                      onChange={(e) => setGuideForm({ ...guideForm, classSize: Number(e.target.value) })}
                      className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-amber-500"
                    />
                  </div>
                </div>

                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                      Phương thức dạy
                    </label>
                    <select
                      value={guideForm.teachingMode}
                      onChange={(e) => setGuideForm({ ...guideForm, teachingMode: e.target.value as any })}
                      className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-amber-500"
                    >
                      <option value="offline">Trực tiếp (Offline)</option>
                      <option value="online">Trực tuyến (Online)</option>
                      <option value="hybrid">Kết hợp (Hybrid)</option>
                    </select>
                  </div>
                  <div>
                    <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                      Trình độ học sinh
                    </label>
                    <select
                      value={guideForm.studentLevel}
                      onChange={(e) => setGuideForm({ ...guideForm, studentLevel: e.target.value as any })}
                      className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800 focus:outline-none focus:border-amber-500"
                    >
                      <option value="beginner">Cơ bản (Beginner)</option>
                      <option value="intermediate">Trung cấp (Intermediate)</option>
                      <option value="advanced">Nâng cao (Advanced)</option>
                    </select>
                  </div>
                </div>

                <button
                  onClick={handleGenerateGuide}
                  disabled={loading || !guideForm.lectureId}
                  className="w-full py-4 bg-amber-500 text-white rounded-xl font-bold hover:bg-amber-600 transition-all flex items-center justify-center gap-2 disabled:opacity-50"
                >
                  {loading ? <Loader2 size={20} className="animate-spin" /> : <Wand2 size={20} />}
                  Tạo Giáo Án với AI
                </button>

                {generatedGuide && (
                  <div className="mt-6 p-6 bg-amber-50 rounded-2xl border border-amber-100">
                    <h3 className="font-bold text-amber-900 mb-4">{generatedGuide.guide?.title}</h3>
                    <div className="space-y-4">
                      <div>
                        <h4 className="font-bold text-amber-800 text-sm uppercase">Mục tiêu:</h4>
                        <ul className="list-disc list-inside text-amber-700">
                          {generatedGuide.guide?.objectives?.map((obj: string, i: number) => (
                            <li key={i}>{obj}</li>
                          ))}
                        </ul>
                      </div>
                      <div>
                        <h4 className="font-bold text-amber-800 text-sm uppercase">Hoạt động:</h4>
                        <ul className="list-disc list-inside text-amber-700">
                          {generatedGuide.guide?.activities?.map((act: any, i: number) => (
                            <li key={i}>{act.name} - {act.duration} phút</li>
                          ))}
                        </ul>
                      </div>
                    </div>
                  </div>
                )}
              </div>
            </div>
          )}

          {/* Exercises Tab */}
          {activeTab === 'exercises' && (
            <div className="space-y-6">
              <div className="flex items-center gap-3 mb-6">
                <div className="p-3 bg-rose-500 rounded-xl text-white">
                  <ClipboardList size={24} />
                </div>
                <div>
                  <h2 className="text-xl font-bold text-gray-900">Tạo Bài Tập</h2>
                  <p className="text-sm text-gray-400">AI sẽ tạo bài tập luyện tập</p>
                </div>
              </div>

              <button
                onClick={handleGenerateExercises}
                disabled={loading || !selectedCourseId}
                className="w-full py-4 bg-rose-500 text-white rounded-xl font-bold hover:bg-rose-600 transition-all flex items-center justify-center gap-2 disabled:opacity-50"
              >
                {loading ? <Loader2 size={20} className="animate-spin" /> : <Wand2 size={20} />}
                Tạo Bài Tập với AI
              </button>

              {generatedExercises && (
                <div className="mt-6 p-6 bg-rose-50 rounded-2xl border border-rose-100">
                  <h3 className="font-bold text-rose-900 mb-4">
                    Đã tạo {generatedExercises.exercises?.length} bài tập
                  </h3>
                  <div className="space-y-2">
                    {generatedExercises.exercises?.map((ex: any, i: number) => (
                      <div key={i} className="p-3 bg-white rounded-lg border border-rose-100">
                        <p className="text-rose-800 font-medium">{ex.question}</p>
                      </div>
                    ))}
                  </div>
                </div>
              )}
            </div>
          )}

          {/* Exam Tab */}
          {activeTab === 'exam' && (
            <div className="space-y-6">
              <div className="flex items-center gap-3 mb-6">
                <div className="p-3 bg-purple-500 rounded-xl text-white">
                  <Award size={24} />
                </div>
                <div>
                  <h2 className="text-xl font-bold text-gray-900">Tạo Đề Thi</h2>
                  <p className="text-sm text-gray-400">AI sẽ tạo đề thi hoàn chỉnh với đáp án</p>
                </div>
              </div>

              <div className="space-y-4">
                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                      Số câu hỏi
                    </label>
                    <input
                      type="number"
                      value={examForm.questionCount}
                      onChange={(e) => setExamForm({ ...examForm, questionCount: Number(e.target.value) })}
                      className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800"
                    />
                  </div>
                  <div>
                    <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                      Độ khó
                    </label>
                    <select
                      value={examForm.difficulty}
                      onChange={(e) => setExamForm({ ...examForm, difficulty: e.target.value as any })}
                      className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800"
                    >
                      <option value="easy">Dễ</option>
                      <option value="medium">Trung bình</option>
                      <option value="hard">Khó</option>
                      <option value="mixed">Hỗn hợp</option>
                    </select>
                  </div>
                </div>

                <div className="flex items-center gap-2">
                  <input
                    type="checkbox"
                    checked={examForm.includeAnswerKey}
                    onChange={(e) => setExamForm({ ...examForm, includeAnswerKey: e.target.checked })}
                    className="w-5 h-5 rounded border-gray-300 text-purple-500 focus:ring-purple-500"
                  />
                  <label className="text-sm font-medium text-gray-700">Bao gồm đáp án</label>
                </div>

                <button
                  onClick={handleGenerateExam}
                  disabled={loading || !selectedCourseId}
                  className="w-full py-4 bg-purple-500 text-white rounded-xl font-bold hover:bg-purple-600 transition-all flex items-center justify-center gap-2 disabled:opacity-50"
                >
                  {loading ? <Loader2 size={20} className="animate-spin" /> : <Wand2 size={20} />}
                  Tạo Đề Thi với AI
                </button>

                {generatedExam && (
                  <div className="mt-6 p-6 bg-purple-50 rounded-2xl border border-purple-100">
                    <h3 className="font-bold text-purple-900 mb-4">{generatedExam.exam?.title}</h3>
                    <p className="text-purple-700 mb-4">{generatedExam.exam?.instructions}</p>
                    <div className="space-y-2">
                      {generatedExam.exam?.questions?.map((q: any, i: number) => (
                        <div key={i} className="p-3 bg-white rounded-lg border border-purple-100">
                          <p className="text-purple-800 font-medium">{i + 1}. {q.content}</p>
                        </div>
                      ))}
                    </div>
                    {generatedExam.answerKey && (
                      <div className="mt-4 p-4 bg-white rounded-lg border border-purple-100">
                        <h4 className="font-bold text-purple-800 mb-2">Đáp án</h4>
                        <pre className="text-sm text-purple-700 whitespace-pre-wrap">
                          {JSON.stringify(generatedExam.answerKey, null, 2)}
                        </pre>
                      </div>
                    )}
                  </div>
                )}
              </div>
            </div>
          )}

          {/* Materials Tab */}
          {activeTab === 'materials' && (
            <div className="space-y-6">
              <div className="flex items-center gap-3 mb-6">
                <div className="p-3 bg-indigo-500 rounded-xl text-white">
                  <Presentation size={24} />
                </div>
                <div>
                  <h2 className="text-xl font-bold text-gray-900">Tạo Tài Liệu Giảng Dạy</h2>
                  <p className="text-sm text-gray-400">AI sẽ tạo slides, handouts, worksheets</p>
                </div>
              </div>

              <div className="space-y-4">
                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    Loại tài liệu
                  </label>
                  <select
                    value={materialsForm.type}
                    onChange={(e) => setMaterialsForm({ ...materialsForm, type: e.target.value as any })}
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800"
                  >
                    <option value="slides">Slides</option>
                    <option value="handout">Handout</option>
                    <option value="worksheet">Worksheet</option>
                    <option value="all">Tất cả</option>
                  </select>
                </div>

                <button
                  onClick={handleGenerateMaterials}
                  disabled={loading || !selectedCourseId}
                  className="w-full py-4 bg-indigo-500 text-white rounded-xl font-bold hover:bg-indigo-600 transition-all flex items-center justify-center gap-2 disabled:opacity-50"
                >
                  {loading ? <Loader2 size={20} className="animate-spin" /> : <Wand2 size={20} />}
                  Tạo Tài Liệu với AI
                </button>

                {generatedMaterials && (
                  <div className="mt-6 p-6 bg-indigo-50 rounded-2xl border border-indigo-100">
                    <h3 className="font-bold text-indigo-900 mb-4">Tài liệu đã tạo</h3>
                    <div className="space-y-3">
                      {generatedMaterials.materials?.slides && (
                        <a
                          href={generatedMaterials.materials.slides.url}
                          target="_blank"
                          rel="noopener noreferrer"
                          className="flex items-center gap-3 p-3 bg-white rounded-xl border border-indigo-100 hover:border-indigo-300 transition-all"
                        >
                          <Presentation size={20} className="text-indigo-500" />
                          <div>
                            <p className="font-medium text-indigo-900">Slides</p>
                            <p className="text-sm text-indigo-600">{generatedMaterials.materials.slides.pages} trang</p>
                          </div>
                          <Download size={18} className="ml-auto text-indigo-400" />
                        </a>
                      )}
                    </div>
                  </div>
                )}
              </div>
            </div>
          )}

          {/* Analytics Tab */}
          {activeTab === 'analytics' && (
            <div className="space-y-6">
              <div className="flex items-center gap-3 mb-6">
                <div className="p-3 bg-cyan-500 rounded-xl text-white">
                  <BarChart3 size={24} />
                </div>
                <div>
                  <h2 className="text-xl font-bold text-gray-900">Phân Tích Khóa Học</h2>
                  <p className="text-sm text-gray-400">AI phân tích chất lượng và hiệu quả khóa học</p>
                </div>
              </div>

              <div className="grid grid-cols-3 gap-4">
                <button
                  onClick={handleAnalyzeQuality}
                  disabled={loading || !selectedCourseId}
                  className="p-4 bg-cyan-50 border border-cyan-100 rounded-xl text-cyan-700 font-bold hover:bg-cyan-100 transition-all flex flex-col items-center gap-2 disabled:opacity-50"
                >
                  <Target size={24} />
                  <span className="text-xs uppercase">Chất Lượng Nội Dung</span>
                </button>

                <button
                  onClick={handleGetAnalytics}
                  disabled={loading || !selectedCourseId}
                  className="p-4 bg-cyan-50 border border-cyan-100 rounded-xl text-cyan-700 font-bold hover:bg-cyan-100 transition-all flex flex-col items-center gap-2 disabled:opacity-50"
                >
                  <TrendingUp size={24} />
                  <span className="text-xs uppercase">Analytics Học Viên</span>
                </button>

                <button
                  onClick={handleAnalyzeDifficulty}
                  disabled={loading || !selectedCourseId}
                  className="p-4 bg-cyan-50 border border-cyan-100 rounded-xl text-cyan-700 font-bold hover:bg-cyan-100 transition-all flex flex-col items-center gap-2 disabled:opacity-50"
                >
                  <Lightbulb size={24} />
                  <span className="text-xs uppercase">Độ Khó</span>
                </button>
              </div>

              {contentQuality && (
                <div className="p-6 bg-cyan-50 rounded-2xl border border-cyan-100">
                  <h3 className="font-bold text-cyan-900 mb-4">Chất lượng nội dung</h3>
                  <div className="grid grid-cols-4 gap-4 mb-4">
                    <div className="text-center p-3 bg-white rounded-xl">
                      <p className="text-2xl font-bold text-cyan-600">{contentQuality.quality?.overall}/10</p>
                      <p className="text-xs text-gray-500 uppercase">Tổng thể</p>
                    </div>
                    <div className="text-center p-3 bg-white rounded-xl">
                      <p className="text-2xl font-bold text-cyan-600">{contentQuality.quality?.clarity}/10</p>
                      <p className="text-xs text-gray-500 uppercase">Rõ ràng</p>
                    </div>
                    <div className="text-center p-3 bg-white rounded-xl">
                      <p className="text-2xl font-bold text-cyan-600">{contentQuality.quality?.completeness}/10</p>
                      <p className="text-xs text-gray-500 uppercase">Đầy đủ</p>
                    </div>
                    <div className="text-center p-3 bg-white rounded-xl">
                      <p className="text-2xl font-bold text-cyan-600">{contentQuality.quality?.engagement}/10</p>
                      <p className="text-xs text-gray-500 uppercase">Hấp dẫn</p>
                    </div>
                  </div>
                  <div>
                    <h4 className="font-bold text-cyan-800 text-sm mb-2">Gợi ý cải thiện:</h4>
                    <ul className="list-disc list-inside text-cyan-700 text-sm">
                      {contentQuality.suggestions?.map((s: string, i: number) => (
                        <li key={i}>{s}</li>
                      ))}
                    </ul>
                  </div>
                </div>
              )}

              {courseAnalytics && (
                <div className="p-6 bg-cyan-50 rounded-2xl border border-cyan-100">
                  <h3 className="font-bold text-cyan-900 mb-4">Analytics khóa học</h3>
                  <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
                    <div className="p-4 bg-white rounded-xl">
                      <Users size={20} className="text-cyan-500 mb-2" />
                      <p className="text-2xl font-bold text-gray-900">{courseAnalytics.overview?.totalStudents}</p>
                      <p className="text-xs text-gray-500 uppercase">Học viên</p>
                    </div>
                    <div className="p-4 bg-white rounded-xl">
                      <TrendingUp size={20} className="text-cyan-500 mb-2" />
                      <p className="text-2xl font-bold text-gray-900">{courseAnalytics.overview?.averageProgress}%</p>
                      <p className="text-xs text-gray-500 uppercase">Tiến độ TB</p>
                    </div>
                    <div className="p-4 bg-white rounded-xl">
                      <Award size={20} className="text-cyan-500 mb-2" />
                      <p className="text-2xl font-bold text-gray-900">{courseAnalytics.overview?.averageScore}</p>
                      <p className="text-xs text-gray-500 uppercase">Điểm TB</p>
                    </div>
                    <div className="p-4 bg-white rounded-xl">
                      <CheckCircle2 size={20} className="text-cyan-500 mb-2" />
                      <p className="text-2xl font-bold text-gray-900">{courseAnalytics.overview?.completionRate}%</p>
                      <p className="text-xs text-gray-500 uppercase">Hoàn thành</p>
                    </div>
                  </div>
                </div>
              )}

              {courseDifficulty && (
                <div className="p-6 bg-cyan-50 rounded-2xl border border-cyan-100">
                  <h3 className="font-bold text-cyan-900 mb-4">Phân tích độ khó</h3>
                  <div className="flex items-center gap-4 mb-4">
                    <div className={`px-4 py-2 rounded-xl font-bold ${
                      courseDifficulty.analysis?.overall === 'easy' ? 'bg-green-100 text-green-700' :
                      courseDifficulty.analysis?.overall === 'moderate' ? 'bg-yellow-100 text-yellow-700' :
                      courseDifficulty.analysis?.overall === 'challenging' ? 'bg-orange-100 text-orange-700' :
                      'bg-red-100 text-red-700'
                    }`}>
                      {courseDifficulty.analysis?.overall === 'easy' ? 'Dễ' :
                       courseDifficulty.analysis?.overall === 'moderate' ? 'Vừa phải' :
                       courseDifficulty.analysis?.overall === 'challenging' ? 'Thử thách' : 'Khó'}
                    </div>
                    <p className="text-cyan-700">{courseDifficulty.analysis?.comparedToAverage}</p>
                  </div>
                </div>
              )}
            </div>
          )}

          {/* Feedback Tab */}
          {activeTab === 'feedback' && (
            <div className="space-y-6">
              <div className="flex items-center gap-3 mb-6">
                <div className="p-3 bg-pink-500 rounded-xl text-white">
                  <MessageSquare size={24} />
                </div>
                <div>
                  <h2 className="text-xl font-bold text-gray-900">Tạo Feedback cho Học Sinh</h2>
                  <p className="text-sm text-gray-400">AI sẽ tạo feedback cá nhân hóa</p>
                </div>
              </div>

              <div className="space-y-4">
                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    ID Học sinh
                  </label>
                  <input
                    value={feedbackForm.studentId}
                    onChange={(e) => setFeedbackForm({ ...feedbackForm, studentId: e.target.value })}
                    placeholder="Nhập ID học sinh"
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800"
                  />
                </div>

                <div>
                  <label className="text-xs font-bold text-gray-400 uppercase tracking-widest block mb-2">
                    Loại feedback
                  </label>
                  <select
                    value={feedbackForm.type}
                    onChange={(e) => setFeedbackForm({ ...feedbackForm, type: e.target.value as any })}
                    className="w-full bg-gray-50 border border-gray-100 rounded-xl px-4 py-3 font-medium text-gray-800"
                  >
                    <option value="general">Tổng quát</option>
                    <option value="specific">Chi tiết</option>
                    <option value="improvement">Cải thiện</option>
                  </select>
                </div>

                <button
                  onClick={handleGenerateFeedback}
                  disabled={loading || !selectedCourseId}
                  className="w-full py-4 bg-pink-500 text-white rounded-xl font-bold hover:bg-pink-600 transition-all flex items-center justify-center gap-2 disabled:opacity-50"
                >
                  {loading ? <Loader2 size={20} className="animate-spin" /> : <Wand2 size={20} />}
                  Tạo Feedback với AI
                </button>

                {studentFeedback && (
                  <div className="mt-6 p-6 bg-pink-50 rounded-2xl border border-pink-100">
                    <h3 className="font-bold text-pink-900 mb-4">Feedback cho học sinh</h3>
                    <div className="space-y-4">
                      <div>
                        <h4 className="font-bold text-pink-800 text-sm uppercase mb-2">Điểm mạnh:</h4>
                        <ul className="list-disc list-inside text-pink-700">
                          {studentFeedback.feedback?.strengths?.map((s: string, i: number) => (
                            <li key={i}>{s}</li>
                          ))}
                        </ul>
                      </div>
                      <div>
                        <h4 className="font-bold text-pink-800 text-sm uppercase mb-2">Cần cải thiện:</h4>
                        <ul className="list-disc list-inside text-pink-700">
                          {studentFeedback.feedback?.improvements?.map((imp: string, i: number) => (
                            <li key={i}>{imp}</li>
                          ))}
                        </ul>
                      </div>
                      <div className="p-4 bg-white rounded-xl border border-pink-100">
                        <p className="text-pink-800 italic">{studentFeedback.feedback?.encouragement}</p>
                      </div>
                    </div>
                  </div>
                )}
              </div>
            </div>
          )}
        </div>
      </div>
    </div>
  );
};

export default AIGeneration;
